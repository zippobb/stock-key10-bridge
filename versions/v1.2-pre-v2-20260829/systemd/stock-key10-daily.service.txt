# /etc/systemd/system/stock-key10-daily.service
[Unit]
Description=Stock Key10 Daily Lightweight Technical Calculation
Wants=network-online.target
After=network-online.target
ConditionPathExists=/opt/stock-automation/scanner/stock_key10_daily.py

[Service]
Type=oneshot
User=root
Group=root
WorkingDirectory=/opt/stock-automation/scanner
EnvironmentFile=-/opt/tushare-mcp/.env
Environment=PYTHONUNBUFFERED=1
ExecStart=/opt/stock-automation/scanner/venv/bin/python /opt/stock-automation/scanner/stock_key10_daily.py
TimeoutStartSec=30min
KillSignal=SIGTERM
StandardOutput=journal
StandardError=journal
SyslogIdentifier=stock-key10-daily

[Install]
WantedBy=multi-user.target
